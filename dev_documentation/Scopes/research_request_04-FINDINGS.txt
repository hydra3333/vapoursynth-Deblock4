#### The Mathematical Logic (Established Facts):
1.  **Field DCT:** The $8 \times 8$ DCT is performed on the $720 \times 288$ individual fields. When you separate an interlaced frame into fields, the vertical block boundaries will fall precisely on row indices that are multiples of 8 ($y \pmod 8 == 0$).

```python
import cv2
import numpy as np


def analyze_dct_characteristics(video_path, num_frames=5):
    cap = cv2.VideoCapture(video_path)
    if not cap.isOpened():
        print(f"Error: Could not open video file {video_path}")
        return

    print(f"Analyzing first {num_frames} frames of: {video_path}\n")

    frame_idx = 0
    while frame_idx < num_frames:
        ret, frame = cap.read()
        if not ret:
            print("End of stream or read error.")
            break

        # Convert to grayscale (luma channel contains the strongest block edges)
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        h, w = gray.shape

        # Separate fields (Top / Even rows and Bottom / Odd rows)
        top_field = gray[0::2, :]
        bottom_field = gray[1::2, :]

        field_h, field_w = top_field.shape

        # We analyze both fields
        for field_name, field in [
            ("Top Field", top_field),
            ("Bottom Field", bottom_field),
        ]:
            # 1. Calculate the mean absolute vertical difference between adjacent rows
            # row_diff[y] = mean(|row[y] - row[y-1]|)
            row_diff = np.mean(
                np.abs(
                    field[1:field_h, :].astype(np.float32)
                    - field[0 : field_h - 1, :].astype(np.float32)
                ),
                axis=1,
            )

            # 2. Compute a localized spike score to isolate sharp block-edge transitions
            # from natural image gradients.
            # spike[y] = diff[y] - mean(neighbors)
            spikes = np.zeros_like(row_diff)
            for y in range(1, len(row_diff) - 1):
                local_bg = (row_diff[y - 1] + row_diff[y + 1]) / 2.0
                spikes[y] = max(0, row_diff[y] - local_bg)

            # 3. Aggregate spikes based on boundary periodicities
            # Note: We skip the top/bottom borders (first and last 16 lines) to avoid
            # head-switching noise or letterbox boundaries.
            border = 16
            mod8_scores = []  # Candidates for Field-DCT boundaries (8, 16, 24...)
            mod4_scores = (
                []
            )  # Candidates for Frame-DCT boundaries (4, 12, 20...) excluding mod 8

            for y in range(border, len(spikes) - border):
                # We align our index with 1-based field line differences
                line_idx = y + 1

                if line_idx % 8 == 0:
                    mod8_scores.append(spikes[y])
                elif line_idx % 4 == 0:
                    mod4_scores.append(spikes[y])

            mean_mod8 = np.mean(mod8_scores) if mod8_scores else 0
            mean_mod4 = np.mean(mod4_scores) if mod4_scores else 0

            # Calculate a ratio
            total = mean_mod8 + mean_mod4
            if total > 0.001:
                field_dct_pct = (mean_mod8 / total) * 100.0
                frame_dct_pct = (mean_mod4 / total) * 100.0
            else:
                field_dct_pct, frame_dct_pct = 0.0, 0.0

            # Interpret the dominant characteristic
            if field_dct_pct > 65.0:
                verdict = "Dominant Field-DCT (Interlaced Coding)"
            elif frame_dct_pct > 65.0:
                verdict = "Dominant Frame-DCT (Progressive/Static Coding)"
            else:
                verdict = "Mixed/Intermediate Coding"

                f"Field-DCT Score: {mean_mod8:.4f} ({field_dct_pct:5.1f}%) | "
                f"Frame-DCT Score: {mean_mod4:.4f} ({frame_dct_pct:5.1f}%) | "
                f"Verdict: {verdict}"
            )
```

If you run this script on a known purely interlaced clip with high motion (which forces the encoder to choose Field-DCT to preserve temporal separation), the `Field-DCT Score` will spike significantly higher than the `Frame-DCT Score`. 

Conversely, running it on static scenes or progressive-scan sequences coded as interlaced (where the encoder chooses Frame-DCT) will cause the `Frame-DCT Score` to dominate.
